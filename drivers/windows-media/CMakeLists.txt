# Build actual IMFActivate + IMFMediaSourceEx + IMFMediaStream2 DLL from pinned
# official source. Camera pipe remains opt-in pending Windows identity/ACL validation.
option(RELEVO_BUILD_CAMERA_PROVIDER "Build pinned COM camera provider" OFF)
option(RELEVO_ENABLE_EXPERIMENTAL_CAMERA_PIPE "Enable unvalidated development camera pipe reader (never production)" OFF)
if(WIN32 AND RELEVO_BUILD_CAMERA_PROVIDER)
 if(NOT MSVC)
  message(FATAL_ERROR "Camera provider baseline requires MSVC and Windows SDK")
 endif()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 set(RELEVO_WIL_INCLUDE "" CACHE PATH "Directory containing wil/cppwinrt.h")
 set(RELEVO_CPPWINRT_INCLUDE "" CACHE PATH "Directory containing winrt/base.h and projections")
 if(NOT EXISTS "${RELEVO_WIL_INCLUDE}/wil/cppwinrt.h" OR NOT EXISTS "${RELEVO_CPPWINRT_INCLUDE}/winrt/base.h")
  message(FATAL_ERROR "Supply WIL and C++/WinRT build-only header directories; see README")
 endif()
 set(camera_source "${CMAKE_CURRENT_BINARY_DIR}/camera-source")
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_camera_source.py"
  --output "${camera_source}" RESULT_VARIABLE prepare_result)
 if(NOT prepare_result EQUAL 0)
  message(FATAL_ERROR "Pinned camera source verification/adaptation failed")
 endif()
 add_library(RelevoCameraSource SHARED
  "${camera_source}/SimpleFrameGenerator.cpp"
  "${camera_source}/SimpleMediaSource.cpp"
  "${camera_source}/SimpleMediaStream.cpp"
  "${camera_source}/VirtualCameraMediaSourceActivate.cpp"
  "${camera_source}/dllmain.cpp"
  "${camera_source}/winrtCommon.cpp"
  "${camera_source}/VirtualCameraMediaSource.def")
 target_include_directories(RelevoCameraSource PRIVATE "${camera_source}" "${RELEVO_WIL_INCLUDE}" "${RELEVO_CPPWINRT_INCLUDE}")
 target_compile_definitions(RelevoCameraSource PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
 if(RELEVO_ENABLE_EXPERIMENTAL_CAMERA_PIPE)
  target_compile_definitions(RelevoCameraSource PRIVATE RELEVO_ENABLE_EXPERIMENTAL_CAMERA_PIPE=1)
 endif()
 target_compile_options(RelevoCameraSource PRIVATE /EHsc /W3)
 target_link_libraries(RelevoCameraSource PRIVATE mfplat mf mfuuid mfsensorgroup windowsapp ole32 propsys cfgmgr32 windowscodecs RelevoCameraDecode)
 set_property(TARGET RelevoCameraSource PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
